cmake_minimum_required(VERSION 3.18)
project(liwetun_service LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "LiWeTun service is Windows-only")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(LIWETUN_TEST_NO_GLOBAL_ROUTES "CI-only: no global routes or system DNS changes" OFF)
option(LIWETUN_TEST_TUN_NETWORK "CI-only: bounded IPv4 routes for interface-bound probes" OFF)
option(LIWETUN_BUILD_PACKAGE_PROBE "CI-only restricted-token package probe" OFF)
option(LIWETUN_TEST_ATTACHED_TUN "CI-only: fence valid traffic against SDK-held attachment" OFF)
option(LIWETUN_TEST_OWNED_PROBE "CI-only: own fixed synthetic socket and require SDK path evidence" OFF)
option(LIWETUN_TEST_OWNED_MIGRATION "CI-only: early-ack owned socket migration negative control" OFF)
option(LIWETUN_TEST_OWNED_BYPASS "CI-only: local reply without SDK-owned flow traversal" OFF)
option(LIWETUN_TEST_OWNED_HTTPS "CI-only: fixed HTTPS exchange over SDK-owned socket" OFF)
if(LIWETUN_TEST_OWNED_HTTPS AND (NOT LIWETUN_TEST_OWNED_PROBE OR LIWETUN_TEST_OWNED_BYPASS))
  message(FATAL_ERROR "Owned HTTPS requires its separate owned-probe fixture")
endif()
if(LIWETUN_TEST_OWNED_BYPASS AND (NOT LIWETUN_TEST_OWNED_PROBE OR LIWETUN_TEST_OWNED_MIGRATION))
  message(FATAL_ERROR "Owned local bypass requires owned probe without migration mode")
endif()
if(LIWETUN_TEST_OWNED_MIGRATION AND NOT LIWETUN_TEST_OWNED_PROBE)
  message(FATAL_ERROR "Owned migration requires the explicit owned probe fixture")
endif()
if(LIWETUN_TEST_OWNED_PROBE AND (NOT LIWETUN_TEST_ATTACHED_TUN OR LIWETUN_TEST_NO_GLOBAL_ROUTES OR LIWETUN_TEST_TUN_NETWORK))
  message(FATAL_ERROR "Owned probe requires attachment guard and ordinary routing in disposable CI")
endif()
if(LIWETUN_TEST_NO_GLOBAL_ROUTES AND LIWETUN_TEST_TUN_NETWORK)
  message(FATAL_ERROR "Conflicting CI network modes")
endif()
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
include("${CMAKE_CURRENT_SOURCE_DIR}/control.cmake")
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../windows" "${CMAKE_CURRENT_BINARY_DIR}/rust")
add_executable(liwetun_service main.cpp backend.cpp)
target_link_libraries(liwetun_service PRIVATE liwetun_control liwetun_native wtsapi32 user32)
if(LIWETUN_TEST_OWNED_PROBE)
  target_compile_definitions(liwetun_service PRIVATE LIWETUN_TEST_OWNED_PROBE=1)
  target_sources(liwetun_service PRIVATE owned_probe_ci.cpp)
  target_link_libraries(liwetun_service PRIVATE ws2_32)
endif()
if(LIWETUN_TEST_OWNED_MIGRATION)
  target_compile_definitions(liwetun_service PRIVATE LIWETUN_TEST_OWNED_MIGRATION=1)
endif()
if(LIWETUN_TEST_OWNED_HTTPS)
  target_compile_definitions(liwetun_service PRIVATE LIWETUN_TEST_OWNED_HTTPS=1)
  target_link_libraries(liwetun_service PRIVATE secur32 crypt32)
endif()
if(LIWETUN_TEST_OWNED_BYPASS)
  target_compile_definitions(liwetun_service PRIVATE LIWETUN_TEST_OWNED_BYPASS=1)
  target_link_libraries(liwetun_service PRIVATE iphlpapi)
endif()
if(LIWETUN_TEST_ATTACHED_TUN)
  target_compile_definitions(liwetun_service PRIVATE LIWETUN_TEST_ATTACHED_TUN=1)
endif()
if(LIWETUN_TEST_NO_GLOBAL_ROUTES)
  target_compile_definitions(liwetun_service PRIVATE LIWETUN_TEST_NO_GLOBAL_ROUTES=1)
endif()
if(LIWETUN_TEST_TUN_NETWORK)
  target_compile_definitions(liwetun_service PRIVATE LIWETUN_TEST_TUN_NETWORK=1)
endif()
add_executable(liwetun_service_probe probe.cpp wfp_probe.cpp)
target_link_libraries(liwetun_service_probe PRIVATE liwetun_control iphlpapi fwpuclnt)
if(LIWETUN_BUILD_PACKAGE_PROBE)
  add_executable(liwetun_package_probe package_probe.cpp)
  target_link_libraries(liwetun_package_probe PRIVATE liwetun_control userenv user32)
  if(MSVC)
    target_compile_options(liwetun_package_probe PRIVATE /W4 /WX /EHsc)
  endif()
endif()
if(MSVC)
  target_compile_options(liwetun_control PRIVATE /W4 /WX /EHsc)
  target_compile_options(liwetun_service PRIVATE /W4 /WX /EHsc)
  target_compile_options(liwetun_service_probe PRIVATE /W4 /WX /EHsc)
endif()
