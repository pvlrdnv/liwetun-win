cmake_minimum_required(VERSION 3.18)
project(liwetun_storage_qualification LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Storage qualification is Windows-only")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include("${CMAKE_CURRENT_SOURCE_DIR}/storage.cmake")
target_compile_definitions(liwetun_storage PUBLIC LIWETUN_STORAGE_TEST_ONLY=1)
add_executable(liwetun_storage_probe probe.cpp)
target_link_libraries(liwetun_storage_probe PRIVATE liwetun_storage)
if(MSVC)
  target_compile_options(liwetun_storage_probe PRIVATE /W4 /WX /EHsc)
endif()
